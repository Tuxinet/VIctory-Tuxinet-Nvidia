+/* VIctory patch: the server is told of section views outside virtual_mutex.
+ * Mapping and unmapping a view used to hold virtual_mutex across a server
+ * round trip (map_view, unmap_view), and every protection change, fault and
+ * query of every thread waited behind it. The server's record of a non-image
+ * view is bookkeeping only; tickets taken under virtual_mutex keep its
+ * requests in the order of the changes they describe. */
+static pthread_mutex_t view_server_mutex = PTHREAD_MUTEX_INITIALIZER;
+static pthread_cond_t view_server_cond = PTHREAD_COND_INITIALIZER;
+static unsigned int view_server_next, view_server_served;
+
+/* virtual_mutex held */
+static unsigned int view_server_ticket(void)
+{
+    return view_server_next++;
+}
+
+static void view_server_wait( unsigned int ticket )
+{
+    pthread_mutex_lock( &view_server_mutex );
+    while (view_server_served != ticket) pthread_cond_wait( &view_server_cond, &view_server_mutex );
+    pthread_mutex_unlock( &view_server_mutex );
+}

+/* VIctory patch: WINE_VICTORY_NO_VIEW_SERVER=1 leaves the server out of
+ * non-image section views altogether (map and unmap alike): its record of
+ * them is bookkeeping, and the round trip was most of what a map costs. */
+static BOOL no_view_server(void)
+{
+    static int no = -1;
+    if (no < 0)
+    {
+        const char *v = getenv( "WINE_VICTORY_NO_VIEW_SERVER" );
+        no = v && v[0] == '1';
+    }
+    return no;
+}
